```python
import torch
from typing import Dict, Tuple

@torch.jit.script
def custom_reward_fn(
    observation: torch.Tensor,
    action: torch.Tensor,
    original_reward: float,
    terminated: bool,
    truncated: bool,
    info: Dict[str, torch.Tensor],
) -> Tuple[torch.Tensor, Dict[str, torch.Tensor]]:
    # Ensure device/dtype consistency
    device = observation.device
    dtype = observation.dtype

    # Convert original env reward to tensor
    original_r = torch.tensor(original_reward, device=device, dtype=dtype)

    # Extract velocity from info (provided by the environment)
    x_vel = info["x_velocity"].to(device=device, dtype=dtype)

    # Parse observation to get torso height (z) and torso angle
    # Walker2d has nq=9, nv=9 => last 9 entries are velocities
    n: int = int(observation.shape[0])
    nv: int = 9
    pos_len: int = n - nv
    vel_start: int = pos_len
    # If current positions are excluded: pos_len == 8 => z at 0, angle at 1
    # If included: pos_len == 9 => z at 1, angle at 2
    z_idx: int = 0
    if pos_len >= 9:
        z_idx = 1
    ang_idx: int = z_idx + 1

    z = observation[z_idx]
    angle = observation[ang_idx]
    qvel = observation[vel_start:]  # length 9

    # Temperatures for transformed components (required named variables)
    temp_fwd = torch.tensor(1.2, device=device, dtype=dtype)       # forward velocity shaping temperature
    temp_posture = torch.tensor(0.3, device=device, dtype=dtype)   # posture shaping temperature

    # Component weights
    w_fwd = torch.tensor(1.0, device=device, dtype=dtype)
    w_posture = torch.tensor(0.5, device=device, dtype=dtype)
    w_alive = torch.tensor(0.3, device=device, dtype=dtype)
    w_original = torch.tensor(0.2, device=device, dtype=dtype)
    ctrl_weight = torch.tensor(2.0e-3, device=device, dtype=dtype)     # control cost coeff
    vel_smooth_weight = torch.tensor(5.0e-4, device=device, dtype=dtype)  # joint velocity smoothness
    term_penalty_weight = torch.tensor(2.0, device=device, dtype=dtype)   # penalty on termination

    # 1) Forward progress (shaped, positive for forward velocity; negative for backward)
    # Use an exponential shaping with temperature and clamping for numerical stability
    fwd_scaled = torch.clamp(x_vel / temp_fwd, min=-5.0, max=5.0)
    fwd_shaped = torch.exp(fwd_scaled) - 1.0
    forward_reward = w_fwd * fwd_shaped

    # 2) Posture reward based on healthy z and angle ranges
    # Healthy ranges from env: z in (0.8, 2.0), angle in (-1.0, 1.0)
    min_z = torch.tensor(0.8, device=device, dtype=dtype)
    max_z = torch.tensor(2.0, device=device, dtype=dtype)
    min_ang = torch.tensor(-1.0, device=device, dtype=dtype)
    max_ang = torch.tensor(1.0, device=device, dtype=dtype)

    z_violation_low = torch.relu(min_z - z)
    z_violation_high = torch.relu(z - max_z)
    ang_violation_low = torch.relu(min_ang - angle)
    ang_violation_high = torch.relu(angle - max_ang)

    posture_violation = z_violation_low + z_violation_high + ang_violation_low + ang_violation_high
    posture_reward_raw = torch.exp(-torch.clamp(posture_violation / temp_posture, min=0.0, max=20.0))
    posture_reward = w_posture * posture_reward_raw

    # 3) Control cost (penalize large torques)
    ctrl_cost = ctrl_weight * torch.sum(action * action)

    # 4) Smoothness cost (penalize large joint velocities)
    # Use mean to keep scale stable
    vel_smooth_cost = vel_smooth_weight * torch.mean(qvel * qvel)

    # 5) Alive bonus and termination penalty
    alive = torch.tensor(0.0 if terminated else 1.0, device=device, dtype=dtype)
    alive_bonus = w_alive * alive
    term_penalty = term_penalty_weight * (1.0 - alive)  # applied only when terminated

    # 6) Original environment reward as auxiliary signal
    orig_component = w_original * original_r

    # Total reward
    total_reward = (
        forward_reward
        + posture_reward
        + alive_bonus
        + orig_component
        - ctrl_cost
        - vel_smooth_cost
        - term_penalty
    )

    # Components dictionary
    components = torch.jit.annotate(Dict[str, torch.Tensor], {})
    components["forward_reward"] = forward_reward
    components["posture_reward"] = posture_reward
    components["alive_bonus"] = alive_bonus
    components["orig_env_reward_scaled"] = orig_component
    components["ctrl_cost"] = -ctrl_cost
    components["vel_smooth_cost"] = -vel_smooth_cost
    components["termination_penalty"] = -term_penalty
    components["x_velocity"] = x_vel
    components["z"] = z
    components["angle"] = angle
    components["total_reward"] = total_reward

    return total_reward, components
```
